\paragraph{WQS 二分 (Aliens trick)}
适用于恰好选 $k$ 个或恰好分 $k$ 段的最优值 $g(k)$, 且 $g$ 关于 $k$ 是凸函数 (求最小值时下凸, 求最大值时上凸). 凸性一般靠打表观察, 或由四边形不等式、费用流模型得到.
\begin{compactitem}
\item \textbf{去掉个数限制}: 每选一个额外加代价 $\lambda$, 做不限个数的 DP, 求 $h(\lambda)=\min_k\{g(k)+\lambda k\}$, 同时记录取到最优时选了几个, 记为 $\mathit{cnt}(\lambda)$.
\item \textbf{打平手时取最少个数}: DP 里值相同的转移一律选个数更少的. 这样 $\mathit{cnt}(\lambda)$ 随 $\lambda$ 单调不增.
\item \textbf{二分}: 找最小的 $\lambda$ 使 $\mathit{cnt}(\lambda)\le k$, 答案是 $g(k)=h(\lambda)-\lambda k$. 注意减去的是 $\lambda k$ 而不是 $\lambda\cdot\mathit{cnt}$, 这样多个 $k$ 共线时也正确.
\item \textbf{求最大值}: 每选一个减去 $\lambda$, $h(\lambda)=\max_k\{g(k)-\lambda k\}$, 同样平手取最少个数, 找最小的 $\lambda$ 使 $\mathit{cnt}(\lambda)\le k$, 答案 $h(\lambda)+\lambda k$.
\item \textbf{细节}: $g$ 是整数时 $\lambda$ 只需整数二分, 范围取相邻差分的最小到最大值 (如 $[-10^{12},10^{12}]$). 总复杂度是 DP 的复杂度乘 $\log V$.
\item \textbf{常见题}: 恰好 $k$ 段划分, 恰好选 $k$ 个不相邻元素, 恰好 $k$ 条白边的最小生成树.
\end{compactitem}

\subsection{调试技巧}
	\begin{itemize}
		\item \verb@signal(SIGSEGV, [](int) { _Exit(0); });@ 把段错误变成 Wrong Answer.
			同理可捕获 SIGABRT (断言失败) 和 SIGFPE (除零).
			\verb@_GLIBCXX_DEBUG@ 报错时产生 SIGABRT (gcc 5.4.0 上似乎是 SIGSEGV).
		\item \verb@feenableexcept(29);@ 在遇到 NaN (\texttt 1), 除零 (\texttt 4), 无穷 (\texttt 8) 和非规格化数 (\texttt{16}) 时终止程序.
	\end{itemize}

\subsection{优化技巧}
	\verb@__builtin_ia32_ldmxcsr(40896);@ 关闭非规格化数 (它们在最小值附近让浮点运算慢 20 倍).
	\subsubsection{位运算技巧}
			\begin{itemize}
			\item \verb@x&-x@ 是 \texttt{x} 的最低位的 1.
			\item \verb@c=x&-x,r=x+c;(((r^x)>>2)/c)|r@ 是比 \texttt{x} 大且 1 的个数相同的下一个数.
			\item 子集和 (SOS DP): \verb@for(int b=0;b<K;b++)for(int i=0;i<(1<<K);i++)@ \\ \verb@  if(i>>b&1)D[i]+=D[i^(1<<b)];@
			\item 枚举所有集合的所有子集, 总共 $3^n$ 次:
		\end{itemize}
\begin{lstlisting}
for(int s=0;s<(1<<n);s++) // 每个集合 s
	for(int t=s;;t=(t-1)&s){ // s 的每个子集 t, 含空集
		// 处理 (s, t)
		if(!t)break;
	}
// 只要非空子集: for(int t=s;t;t=(t-1)&s)
\end{lstlisting}
