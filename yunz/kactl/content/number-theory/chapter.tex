	\paragraph{Bézout 定理}
	对 $a \neq 0$, $b \neq 0$, $d=gcd(a,b)$ 是使下式有整数解的最小正整数
	$$ax+by=d$$
	若 $(x,y)$ 是一组解, 则全部解为
	$$\left(x+\frac{kb}{\gcd(a,b)}, y-\frac{ka}{\gcd(a,b)}\right), \quad k\in\mathbb{Z}$$

	\paragraph{欧拉函数公式}
		$\varphi(n)$ 为 $1\dots n$ 中与 $n$ 互质的数的个数.
		\begin{compactitem}
		\item 通项: $n=\prod p_i^{k_i}$ 时 $\varphi(n)=n\prod_{p\mid n}\left(1-\frac1p\right)$, $\varphi(p^k)=(p-1)p^{k-1}$, $\varphi(1)=1$.
		\item 积性: $\gcd(m,n)=1$ 时 $\varphi(mn)=\varphi(m)\varphi(n)$. 线性筛里: $p\mid i$ 时 $\varphi(ip)=\varphi(i)\cdot p$, 否则 $\varphi(ip)=\varphi(i)(p-1)$.
		\item $\sum_{d\mid n}\varphi(d)=n$; $n>1$ 时与 $n$ 互质的数之和为 $n\varphi(n)/2$.
		\item $\sum_{i=1}^{n}\gcd(i,n)=\sum_{d\mid n}d\cdot\varphi(n/d)$; $\#\{1\le i,j\le n:\gcd(i,j)=1\}=2\sum_{i=1}^{n}\varphi(i)-1$.
		\item 欧拉定理: $\gcd(a,m)=1$ 时 $a^{\varphi(m)}\equiv1\pmod m$, 所以 $a^b\equiv a^{b\bmod\varphi(m)}$. 质数 $p$ 时即费马小定理 $a^{p-1}\equiv1$.
		\item 扩展欧拉定理 (不要求互质): $b\ge\varphi(m)$ 时 $a^b\equiv a^{b\bmod\varphi(m)+\varphi(m)}\pmod m$; $b<\varphi(m)$ 时直接算. 用于指数巨大或幂塔 $a^{a^{a\cdots}}$, 递归 $m\to\varphi(m)$ 约 $O(\log m)$ 层到 1.
		\end{compactitem}

\subsubsection{勾股数}
 勾股数可唯一地表示为
 \[ a=k\cdot (m^{2}-n^{2}),\ \,b=k\cdot (2mn),\ \,c=k\cdot (m^{2}+n^{2}), \]
 其中 $m > n > 0$, $k > 0$, $m \bot n$, 且 $m$ 或 $n$ 为偶数.

\subsubsection{Mobius 函数}
\[
	\mu(n) = \begin{cases} 0 & n \textrm{ is not square free}\\ 1 & n \textrm{ has even number of prime factors}\\ -1 & n \textrm{ has odd number of prime factors}\\\end{cases}
\]
  Mobius 反演:
  \[ g(n) = \sum_{d|n} f(d) \Leftrightarrow f(n) = \sum_{d|n} \mu(d)g(n/d) \]
  其他常用形式:

  $ \sum_{d | n} \mu(d) = [ n = 1] $ (很有用)

  $ g(n) = \sum_{n|d} f(d) \Leftrightarrow f(n) = \sum_{n|d} \mu(d/n)g(d)$

 $ g(n) = \sum_{1 \leq m \leq n} f(\left\lfloor\frac{n}{m}\right \rfloor ) \Leftrightarrow f(n) = \sum_{1\leq m\leq n} \mu(m)g(\left\lfloor\frac{n}{m}\right\rfloor)$
