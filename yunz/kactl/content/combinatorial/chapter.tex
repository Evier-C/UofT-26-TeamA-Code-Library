	\paragraph{环}
		设 $g_S(n)$ 为所有环长都属于集合 $S$ 的 $n$ 元排列个数, 则
		$$\sum_{n=0} ^\infty g_S(n) \frac{x^n}{n!} = \exp\left(\sum_{n\in S} \frac{x^n} {n} \right)$$

	\paragraph{错排}
		没有任何元素在原位置上的排列.
		\[ \mkern-2mu D(n) = (n-1)(D(n-1)+D(n-2)) = n D(n-1)+(-1)^n = \left\lfloor\frac{n!}{e}\right\rceil \]

	\paragraph{Burnside 引理}
		给定对称群 $G$ 和集合 $X$, $X$ 中\emph{在对称意义下}不同的元素个数为
		 \[ {\frac {1}{|G|}}\sum _{{g\in G}}|X^{g}|, \]
		 其中 $X^{g}$ 是被 $g$ 固定的元素 ($g.x = x$).

		 若 $f(n)$ 统计长度为 $n$ 的某种方案数, 取 $G = \mathbb Z_n$ 即可不计旋转对称, 得
		 \[ g(n) = \frac 1 n \sum_{k=0}^{n-1}{f(\text{gcd}(n, k))} = \frac 1 n \sum_{k|n}{f(k)\phi(n/k)}. \]

\subsubsection{划分与子集}
	\paragraph{划分函数}
		把 $n$ 写成若干正整数之和的方案数, 不计顺序.
		\[ p(0) = 1,\ p(n) = \sum_{k \in \mathbb Z \setminus \{0\}}{(-1)^{k+1} p(n - k(3k-1) / 2)} \]
		\[ p(n) \sim 0.145 / n \cdot \exp(2.56 \sqrt{n}) \]

	\paragraph{Lucas 定理}
		设 $n,m$ 为非负整数, $p$ 为素数. 写成 $n=n_kp^k+...+n_1p+n_0$, $m=m_kp^k+...+m_1p+m_0$, 则 $\binom{n}{m} \equiv \prod_{i=0}^k\binom{n_i}{m_i} \pmod{p}$.

\subsubsection{容斥与二项式反演}
	\paragraph{容斥原理}
		\[ \Big|\bigcup_{i=1}^n A_i\Big| = \sum_{\emptyset\ne S\subseteq[n]} (-1)^{|S|+1} \Big|\bigcap_{i\in S} A_i\Big| \]
		\[ \#\{\text{一个条件都不满足}\} = \sum_{S\subseteq[n]} (-1)^{|S|} \Big|\bigcap_{i\in S} A_i\Big| \]
		空集的交是全集. 套路: 条件 $i$ 是坏事件, 钦定一个集合 $S$ 里的坏事件一定发生、其余随意, 算方案数后按 $|S|$ 的奇偶加减. 只和 $|S|$ 有关时写成 $\sum_k (-1)^k \binom{n}{k} h(k)$.
		\begin{compactitem}
		\item 满射 ($n$ 个不同的球放进 $k$ 个不同的盒子且不空): $\sum_{j=0}^{k}(-1)^j\binom{k}{j}(k-j)^n$.
		\item 带上界的插板: $x_1+\dots+x_k=n$, $0\le x_i\le c$ 的解数 $\sum_{j}(-1)^j\binom{k}{j}\binom{n-j(c+1)+k-1}{k-1}$.
		\item 互质计数: $\#\{1\le a,b\le n: \gcd(a,b)=1\} = \sum_{d}\mu(d)\lfloor n/d\rfloor^2$, 莫比乌斯函数就是对质因子做容斥.
		\end{compactitem}
	\paragraph{二项式反演}
		\[ f(n)=\sum_{i=0}^{n}\binom{n}{i}g(i) \iff g(n)=\sum_{i=0}^{n}(-1)^{n-i}\binom{n}{i}f(i) \]
		\[ f(k)=\sum_{i=k}^{n}\binom{i}{k}g(i) \iff g(k)=\sum_{i=k}^{n}(-1)^{i-k}\binom{i}{k}f(i) \]
		第二个是最常用的 至少 转 恰好: 设 $g(k)$ 为恰好满足 $k$ 个条件的方案数, $f(k)$ 为先钦定 $k$ 个条件满足、其余随意的方案数之和 (同一方案会被算 $\binom{i}{k}$ 次), 通常 $f$ 好算, 再反演得 $g$.
		例: 恰好 $k$ 个不动点的排列数, 钦定 $i$ 个不动点 $f(i)=\binom{n}{i}(n-i)!$, 代入上式; $k=0$ 即错排.
	\paragraph{子集反演}
		\[ f(S)=\sum_{T\subseteq S}g(T) \iff g(S)=\sum_{T\subseteq S}(-1)^{|S|-|T|}f(T) \]
		\[ f(S)=\sum_{T\supseteq S}g(T) \iff g(S)=\sum_{T\supseteq S}(-1)^{|T|-|S|}f(T) \]
		对状压可以用子集和 (SOS DP) 的逆运算在 $O(2^n n)$ 内求出. 期望里的 min-max 容斥见 期望与概率技巧 一节.

\subsubsection{计数常用结论}
	\paragraph{插板法与二项式恒等式}
		\begin{compactitem}
		\item $x_1+\dots+x_k=n$ 的非负整数解有 $\binom{n+k-1}{k-1}$ 个, 正整数解有 $\binom{n-1}{k-1}$ 个; 有上界时用上一节的容斥.
		\item 范德蒙德卷积 $\sum_i\binom{n}{i}\binom{m}{k-i}=\binom{n+m}{k}$; 曲棍球棒 $\sum_{i=r}^{n}\binom{i}{r}=\binom{n+1}{r+1}$; 吸收 $k\binom{n}{k}=n\binom{n-1}{k-1}$; $\sum_i\binom{n}{i}^2=\binom{2n}{n}$.
		\end{compactitem}
	\paragraph{球盒模型}
		$n$ 个球放进 $k$ 个盒子, $S(n,k)$ 为第二类 Stirling 数, $p_k(n)$ 为把 $n$ 拆成至多 $k$ 个正整数之和的方案数 ($p_k(n)=p_{k-1}(n)+p_k(n-k)$).
		\begin{center}
		\begin{tabular}{l|c|c}
		 & 允许空盒 & 不允许空盒 \\ \hline
		球不同, 盒不同 & $k^n$ & $k!\,S(n,k)$ \\
		球不同, 盒相同 & $\sum_{i=0}^{k}S(n,i)$ & $S(n,k)$ \\
		球相同, 盒不同 & $\binom{n+k-1}{k-1}$ & $\binom{n-1}{k-1}$ \\
		球相同, 盒相同 & $p_k(n)$ & $p_k(n-k)$ \\
		\end{tabular}
		\end{center}
	\paragraph{反射原理}
		从 $(0,0)$ 每步向右或向上走到 $(n,m)$ 共 $\binom{n+m}{m}$ 种. $n\ge m$ 时始终满足 $y\le x$ (不越过对角线) 的走法有 $\binom{n+m}{m}-\binom{n+m}{m-1}$ 种: 越过对角线的路径在第一次碰到 $y=x+1$ 之后, 把后半段关于这条线翻折, 一一对应到从 $(0,0)$ 走到 $(m-1,n+1)$ 的路径. $n=m$ 时就是 Catalan 数.
	\paragraph{Pólya 计数 (项链)}
		长为 $n$ 的环用 $k$ 种颜色染色, 旋转后相同算一种 (项链): $N=\frac1n\sum_{d\mid n}\varphi(d)\,k^{n/d}$.
		翻转后相同也算一种 (手镯): $n$ 为奇数时 $\frac12\big(N+k^{(n+1)/2}\big)$, $n$ 为偶数时 $\frac12\big(N+\frac12(k^{n/2}+k^{n/2+1})\big)$.
		一般地, 方案数 $=\frac{1}{|G|}\sum_{g\in G}k^{c(g)}$, $c(g)$ 是置换 $g$ 的环数 (即 Burnside 引理).
	\paragraph{LGV 引理}
		起点 $a_1..a_k$, 终点 $b_1..b_k$, 记 $e(a,b)$ 为 $a$ 到 $b$ 的路径数. 若互不相交的路径组只能让 $a_i$ 连到 $b_i$ (如网格上起点、终点都按同一方向排好序), 则 $k$ 条两两没有公共点的路径的方案数等于行列式 $\det\big[e(a_i,b_j)\big]_{k\times k}$. 网格上 $e=\binom{\Delta x+\Delta y}{\Delta x}$, 行列式用 \texttt{IntDeterminant.h} 取模计算.
	\paragraph{其他结论}
		\begin{compactitem}
		\item \textbf{Pick 定理}: 顶点都是格点的简单多边形, 面积 $A=I+\frac{B}{2}-1$, $I$ 为内部格点数, $B$ 为边上格点数, 一条边上的格点数是 $\gcd(|\Delta x|,|\Delta y|)$.
		\item \textbf{欧拉公式}: 平面图 $V-E+F=1+C$ ($F$ 含外面的无穷面, $C$ 为连通块数), 连通时 $V-E+F=2$. 常用于数平面被分成几块.
		\item \textbf{有标号有根树} $n^{n-1}$ 棵; 有标号森林, $k$ 棵树且 $1..k$ 分别在不同的树里 (作为根): $k\,n^{n-k-1}$ 种.
		\item \textbf{Prüfer 序列}: 有标号无根树与长 $n-2$ 的序列一一对应, 每个点在序列中出现 $\deg-1$ 次, 由此得到无根树 $n^{n-2}$ 棵和按度数计数的公式.
		\end{compactitem}

\subsubsection{常用计数数列}
	\paragraph{Bernoulli 数}
		Bernoulli 数的 EGF 为 $B(t)=\frac{t}{e^t-1}$ (可用 FFT 求).
		$B[0,\ldots] = [1, -\frac{1}{2}, \frac{1}{6}, 0, -\frac{1}{30}, 0, \frac{1}{42}, \ldots]$

		幂和:
		\small
		\[ \sum_{i=1}^n n^m = \frac{1}{m+1} \sum_{k=0}^m \binom{m+1}{k} B_k \cdot (n+1)^{m+1-k} \]
		\normalsize

		无穷和的 Euler-Maclaurin 公式:
		\small
		\[ \sum_{i=m}^{\infty} f(i) = \int_m^\infty f(x) dx - \sum_{k=1}^\infty \frac{B_k}{k!}f^{(k-1)}(m) \]
		\[ \approx \int_{m}^\infty f(x)dx + \frac{f(m)}{2} - \frac{f'(m)}{12} + \frac{f'''(m)}{720} + O(f^{(5)}(m)) \]
		\normalsize

	\paragraph{第一类 Stirling 数}
		$n$ 元排列中恰有 $k$ 个环的排列数.
		\begin{align*}
			&c(n,k) = c(n-1,k-1) + (n-1) c(n-1,k),\ c(0,0) = 1 \\
			&\textstyle \sum_{k=0}^n c(n,k)x^k = x(x+1) \dots (x+n-1)
		\end{align*}
		$c(8,k) = 8, 0, 5040, 13068, 13132, 6769, 1960, 322, 28, 1$ \\
		$c(n,2) = 0, 0, 1, 3, 11, 50, 274, 1764, 13068, 109584, \dots$

	\paragraph{Eulerian 数}
		恰有 $k$ 个元素大于前一个元素的排列 $\pi \in S_n$ 个数. 等价于: 恰有 $k$ 个 $j$ 使 $\pi(j)>\pi(j+1)$; 恰有 $k+1$ 个 $j$ 使 $\pi(j)\geq j$; 恰有 $k$ 个 $j$ 使 $\pi(j)>j$.
		$$E(n,k) = (n-k)E(n-1,k-1) + (k+1)E(n-1,k)$$
		$$E(n,0) = E(n,n-1) = 1$$
		$$E(n,k) = \sum_{j=0}^k(-1)^j\binom{n+1}{j}(k+1-j)^n$$

	\paragraph{第二类 Stirling 数}
		把 $n$ 个不同元素恰好分成 $k$ 组的方案数.
		$$S(n,k) = S(n-1,k-1) + k S(n-1,k)$$
		$$S(n,1) = S(n,n) = 1$$
		$$S(n,k) = \frac{1}{k!}\sum_{j=0}^k (-1)^{k-j}\binom{k}{j}j^n$$

	\paragraph{Bell 数}
		$n$ 个不同元素的划分总数. $B(n) =$
		$1, 1, 2, 5, 15, 52, 203, 877, 4140, 21147, \dots$. 对素数 $p$,
		\[ B(p^m+n)\equiv mB(n)+B(n+1) \pmod{p} \]

	\paragraph{有标号无根树}
		\# $n$ 个点: $n^{n-2}$ \\
		\# 连接 $k$ 棵已有的树 (大小为 $n_i$): $n_1n_2\cdots n_k n^{k-2}$ \\
		\# 度数为 $d_i$: $(n-2)! / ((d_1-1)! \cdots (d_n-1)!)$

	\paragraph{Catalan 数}
		\[ C_n=\frac{1}{n+1}\binom{2n}{n}= \binom{2n}{n}-\binom{2n}{n+1} = \frac{(2n)!}{(n+1)!n!} \]
		\[ C_0=1,\ C_{n+1} = \frac{2(2n+1)}{n+2}C_n,\ C_{n+1}=\sum C_iC_{n-i} \]
		${C_n = 1, 1, 2, 5, 14, 42, 132, 429, 1430, 4862, 16796, 58786, \dots}$
		\begin{itemize}[noitemsep]
			\item $n\times n$ 网格中不越过对角线的单调路径.
			\item $n$ 对括号组成的合法括号串.
			\item 有 $n+1$ 个叶子的二叉树 (每点 0 或 2 个儿子).
			\item 有 $n+1$ 个点的有序树.
			\item 连对角线把凸 $n+2$ 边形剖分成三角形的方案.
			\item 不含长为 3 的递增子序列的 $[n]$ 的排列.
		\end{itemize}
